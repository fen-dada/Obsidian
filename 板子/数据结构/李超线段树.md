动态开点李超线段树
清空：tot = rt = 0;
空间：直线O(Q)，线段O(QlogV)
斜率优化时插入{-k,-b}
```cpp
const int N = 2e5 + 5; // 只插全局直线，开到操作次数即可
const ll INF = 4e18;

struct Line {
    ll k, b;
    ll eval(ll x) const { return k * x + b; }
} tr[N];
int lc[N], rc[N], tot, rt;

void insert(int &p, ll l, ll r, Line f) {
    if (!p) {
        p = ++tot;
        lc[p] = rc[p] = 0;
        tr[p] = f;
        return;
    }
    ll mid = l + ((r - l) >> 1);
    if (f.eval(mid) > tr[p].eval(mid))
        swap(tr[p], f);
    if (l == r)
        return;
    if (f.eval(l) > tr[p].eval(l))
        insert(lc[p], l, mid, f);
    else if (f.eval(r) > tr[p].eval(r))
        insert(rc[p], mid + 1, r, f);
}

ll query(int p, ll l, ll r, ll x) {
    if (!p)
        return -INF;
    ll res = tr[p].eval(x);
    if (l == r)
        return res;
    ll mid = l + ((r - l) >> 1);
    return max(res, x <= mid ? query(lc[p], l, mid, x)
                             : query(rc[p], mid + 1, r, x));
}
```
插入线段：
```cpp
void add_seg(int &p, ll l, ll r, ll ql, ll qr, Line f) {
    if (ql <= l && r <= qr) {
        insert(p, l, r, f);
        return;
    }
    if (!p) {
        p = ++tot;
        lc[p] = rc[p] = 0;
        tr[p] = {0, -INF};
    }
    ll mid = l + ((r - l) >> 1);
    if (ql <= mid)
        add_seg(lc[p], l, mid, ql, qr, f);
    if (qr > mid)
        add_seg(rc[p], mid + 1, r, ql, qr, f);
}
```
