一张无向图的生成树数量，等于它的拉普拉斯矩阵删去任意一个点对应的行、列后，剩余矩阵的行列式。
在枚举最小生成树的边权w时，w的数量固定，把不为w的边缩点以后，每一层权值的最小生成树数量互相独立，满足乘法定理。
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl "\n"
struct edge {
    int u, v, w;
    bool operator<(const edge &p) const { return w < p.w; }
};
const int mod = 31011;
const int MOD = 31011;
struct DSU {
    std::vector<int> f, siz;

    DSU() {}
    DSU(int n) { init(n + 1); }

    void init(int n) {
        f.resize(n);
        std::iota(f.begin(), f.end(), 0);
        siz.assign(n, 1);
    }

    int find(int x) {
        while (x != f[x]) {
            x = f[x] = f[f[x]];
        }
        return x;
    }

    bool same(int x, int y) { return find(x) == find(y); }

    bool merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) {
            return false;
        }
        siz[x] += siz[y];
        f[y] = x;
        return true;
    }

    int size(int x) { return siz[find(x)]; }
};
ll det(vector<vector<ll>> a) {
    int n = a.size();

    // 把所有元素调整到 [0, MOD)。
    for (auto &row : a) {
        for (ll &x : row) {
            x = (x % MOD + MOD) % MOD;
        }
    }

    ll ans = 1;

    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            while (a[j][i] != 0) {
                ll t = a[i][i] / a[j][i];

                for (int k = i; k < n; ++k) {
                    a[i][k] = (a[i][k] - t * a[j][k] % MOD + MOD) % MOD;
                }

                swap(a[i], a[j]);
                ans = (MOD - ans) % MOD;
            }
        }

        if (a[i][i] == 0)
            return 0;

        ans = ans * a[i][i] % MOD;
    }

    return ans;
}
void solve() {
    int n, m;
    cin >> n >> m;
    vector<edge> p;
    for (int i = 1; i <= m; ++i) {
        int x, y, w;
        cin >> x >> y >> w;
        p.push_back({x, y, w});
    }
    sort(p.begin(), p.end());
    DSU dsu(n + 1);
    vector<edge> tree;
    for (auto [x, y, w] : p) {
        if (dsu.merge(x, y)) {
            tree.push_back({x, y, w});
        }
    }
    if ((int)tree.size() != n - 1) {
        cout << 0 << '\n';
        return;
    }
    ll ans = 1;
    for (int l = 0; l < m;) {
        int r = l + 1;
        while (r < m && p[r].w == p[l].w)
            r++;
        int tot = r - l;
        int need = 0;
        DSU base(n + 1);
        for (auto [u, v, w] : tree) {
            if (w != p[l].w) {
                base.merge(u, v);
            } else {
                ++need;
            }
        }
        if (need != 0) {
            int cnt = 0;
            vector<int> id(n + 1, -1);
            for (int i = 1; i <= n; ++i) {
                int f = base.find(i);
                if (id[f] == -1) {
                    id[f] = cnt++;
                }
            }
            vector<vector<ll>> L(cnt, vector<ll>(cnt));
            for (int i = l; i < r; ++i) {
                auto [u, v, w] = p[i];
                u = id[base.find(u)];
                v = id[base.find(v)];
                if (u == v)
                    continue;
                ++L[u][u];
                ++L[v][v];
                --L[u][v];
                --L[v][u];
            }
            vector<vector<ll>> a(cnt - 1, vector<ll>(cnt - 1));
            for (int i = 0; i < cnt - 1; ++i) {
                for (int j = 0; j < cnt - 1; ++j) {
                    a[i][j] = L[i][j];
                }
            }
            ll ways = det(a);
            if (ways) {
                ans = ans * ways % mod;
            }
        }
        l = r;
    }
    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int T = 1;
    while (T--) {
        solve();
    }
}
```
