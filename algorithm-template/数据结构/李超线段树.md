```cpp
// 动态开点李超线段树（可直接复制）
// IS_MIN = false 维护最大值；IS_MIN = true 维护最小值
template <bool IS_MIN = false> struct LiChaoTree {
    using i64 = long long;
    static constexpr i64 INF = 4e18;
    static constexpr i64 ID_VAL = IS_MIN ? INF : -INF;

    struct Line {
        i64 k, b;
        i64 eval(i64 x) const { return k * x + b; }
        // 若 k * x 可能爆 long long，把 eval 改成：
        // __int128 eval(i64 x) const { return (__int128)k * x + b; }
    };

    struct Node {
        Line f;
        int32_t lc = 0, rc = 0;
    };

    i64 L, R;
    int32_t root = 0;
    vector<Node> tr;

    LiChaoTree(i64 L = -1e9, i64 R = 1e9, int32_t reserve_sz = 0) : L(L), R(R) {
        if (reserve_sz > 0)
            tr.reserve(reserve_sz + 1);
        tr.push_back({{0, ID_VAL}, 0, 0}); // 0 号作空节点
    }

    // O(1) 多测清空（不释放已申请的内存）
    void clear() {
        tr.resize(1);
        root = 0;
    }

    static bool better(const Line &a, const Line &b, i64 x) {
        return IS_MIN ? (a.eval(x) < b.eval(x)) : (a.eval(x) > b.eval(x));
    }

    int32_t new_node(Line f) {
        tr.push_back({f, 0, 0});
        return (int32_t)tr.size() - 1;
    }

    int32_t push_line(int32_t p, i64 l, i64 r, Line f) {
        if (!p)
            return new_node(f);
        i64 mid = l + ((r - l) >> 1);
        if (better(f, tr[p].f, mid))
            swap(tr[p].f, f);
        if (l == r)
            return p;
        if (better(f, tr[p].f, l)) {
            int32_t ch = push_line(tr[p].lc, l, mid, f);
            tr[p].lc = ch;
        } else if (better(f, tr[p].f, r)) {
            int32_t ch = push_line(tr[p].rc, mid + 1, r, f);
            tr[p].rc = ch;
        }
        return p;
    }

    int32_t add_seg(int32_t p, i64 l, i64 r, i64 ql, i64 qr, Line f) {
        if (ql <= l && r <= qr)
            return push_line(p, l, r, f);
        if (!p)
            p = new_node({0, ID_VAL});
        i64 mid = l + ((r - l) >> 1);
        if (ql <= mid) {
            int32_t ch = add_seg(tr[p].lc, l, mid, ql, qr, f);
            tr[p].lc = ch;
        }
        if (qr > mid) {
            int32_t ch = add_seg(tr[p].rc, mid + 1, r, ql, qr, f);
            tr[p].rc = ch;
        }
        return p;
    }

    i64 query(int32_t p, i64 l, i64 r, i64 x) const {
        if (!p)
            return ID_VAL;
        i64 res = tr[p].f.eval(x);
        if (l == r)
            return res;
        i64 mid = l + ((r - l) >> 1);
        i64 sub = (x <= mid) ? query(tr[p].lc, l, mid, x)
                             : query(tr[p].rc, mid + 1, r, x);
        return IS_MIN ? min(res, sub) : max(res, sub);
    }

    int32_t merge(int32_t u, int32_t v, i64 l, i64 r) {
        if (!u || !v)
            return u | v;
        if (l == r) {
            if (better(tr[v].f, tr[u].f, l))
                tr[u].f = tr[v].f;
            return u;
        }
        i64 mid = l + ((r - l) >> 1);
        int32_t lch = merge(tr[u].lc, tr[v].lc, l, mid);
        int32_t rch = merge(tr[u].rc, tr[v].rc, mid + 1, r);
        tr[u].lc = lch;
        tr[u].rc = rch;
        return push_line(u, l, r, tr[v].f);
    }

    // ---------- 1. 单树便捷接口（斜率优化 / 常规题直接调） ----------
    void add_line(i64 k, i64 b) { root = push_line(root, L, R, {k, b}); }

    void add_segment(i64 ql, i64 qr, i64 k, i64 b) {
        ql = max(ql, L), qr = min(qr, R);
        if (ql <= qr)
            root = add_seg(root, L, R, ql, qr, {k, b});
    }

    i64 query(i64 x) const { return query(root, L, R, x); }

    // ---------- 2. 多树接口（树形 DP 李超树合并） ----------
    template <class I> void add_line(I &rt, i64 k, i64 b) {
        rt = push_line((int32_t)rt, L, R, {k, b});
    }

    template <class I> void add_segment(I &rt, i64 ql, i64 qr, i64 k, i64 b) {
        ql = max(ql, L), qr = min(qr, R);
        if (ql <= qr)
            rt = add_seg((int32_t)rt, L, R, ql, qr, {k, b});
    }

    i64 query(int32_t rt, i64 x) const { return query(rt, L, R, x); }

    int32_t merge(int32_t u, int32_t v) { return merge(u, v, L, R); }
};
```
